%% ── Contexte ────────────────────────────────────────────────────────
\begin{tcolorbox}[sectionbox, title={\faQuestion\quad Contexte \& Problématique}]
Les déploiements traditionnels souffrent de trois limites majeures :
\vspace{4pt}

\bitem[red]{Portabilité : dépendances liées à la machine hôte}
\bitem[red]{Scalabilité : provisioning manuel, lent et coûteux}
\bitem[red]{Reproductibilité : déploiements fragiles, rollback difficile}

\vspace{6pt}
\textbf{Solution adoptée :}\\[4pt]
\bitem{Conteneurisation Docker → images immuables et portables}
\bitem{Kubernetes → orchestration automatique, scaling, self-healing}
\bitem{GKE Autopilot → infrastructure managée, zéro ops nœuds}
\end{tcolorbox}

%% ── Docker ──────────────────────────────────────────────────────────
\begin{tcolorbox}[sectionbox, title={\faDocker\quad Conteneurisation Docker}]

\textbf{Stratégie multi-stage build} : l'image de production ne contient que le runtime (pas les outils de build), ce qui réduit la taille et la surface d'attaque.

\vspace{8pt}
\textbf{Dockerfile Backend} (\texttt{backend/Dockerfile}) :
\begin{tcolorbox}[codebox]
\begin{lstlisting}
# Stage 1 : Build Maven
FROM maven:3.9-eclipse-temurin-21 AS build
WORKDIR /app
COPY pom.xml .
RUN mvn dependency:go-offline -B
COPY src ./src
RUN mvn package -DskipTests -B

# Stage 2 : Runtime leger (~90 MB)
FROM eclipse-temurin:21-jre-alpine
RUN addgroup -S spring \
 && adduser -S spring -G spring
COPY --from=build /app/target/*.jar app.jar
USER spring
EXPOSE 8080
HEALTHCHECK --interval=30s --timeout=5s \
  CMD wget -qO- localhost:8080/actuator/health
ENTRYPOINT ["java", "-jar", "app.jar"]
\end{lstlisting}
\end{tcolorbox}

\vspace{6pt}
\textbf{Dockerfile Frontend} (\texttt{frontend/Dockerfile}) :
\begin{tcolorbox}[codebox]
\begin{lstlisting}
# Stage 1 : Build Angular (prod)
FROM node:20-alpine AS build
WORKDIR /app
COPY package.json package-lock.json ./
RUN npm ci --ignore-scripts
COPY . .
RUN npm run build -- --configuration production

# Stage 2 : Serveur NGINX
FROM nginx:alpine
COPY --from=build \
  /app/dist/frontend/browser \
  /usr/share/nginx/html
COPY nginx.conf /etc/nginx/conf.d/default.conf
EXPOSE 80
CMD ["nginx", "-g", "daemon off;"]
\end{lstlisting}
\end{tcolorbox}

\vspace{6pt}
\textbf{NGINX} joue un double rôle : serveur de fichiers Angular \textbf{ET} reverse proxy vers le backend :
\begin{tcolorbox}[codebox]
\begin{lstlisting}
location /api/ {
  # Proxy vers le service K8s backend
  proxy_pass http://backend:8080/api/;
}
location / {
  # SPA routing Angular
  try_files $uri $uri/ /index.html;
}
\end{lstlisting}
\end{tcolorbox}

\vspace{6pt}
\textbf{Build \& Push :}
\begin{tcolorbox}[codebox]
\begin{lstlisting}
$ docker build -t raybob2001/user-role-\
  management-backend:latest ./backend

$ docker push raybob2001/user-role-\
  management-backend:latest
latest: digest: sha256:a1b2c3...
\end{lstlisting}
\end{tcolorbox}

\end{tcolorbox}

%% ── Kubernetes ──────────────────────────────────────────────────────
\begin{tcolorbox}[sectionbox, title={\faServer\quad Orchestration Kubernetes}]

\textbf{Backend Deployment} — 2 réplicas avec readiness probe :
\begin{tcolorbox}[codebox]
\begin{lstlisting}
spec:
  replicas: 2
  containers:
    - name: backend
      image: raybob2001/...-backend:latest
      resources:
        limits:
          cpu: 500m
          memory: 512Mi
      envFrom:
        - configMapRef:
            name: app-config
        - secretRef:
            name: app-secret
      readinessProbe:
        httpGet:
          path: /actuator/health
          port: 8080
        initialDelaySeconds: 15
\end{lstlisting}
\end{tcolorbox}

\vspace{6pt}
\textbf{PostgreSQL} avec stockage persistant (PVC 1Gi) :
\begin{tcolorbox}[codebox]
\begin{lstlisting}
volumes:
  - name: postgres-data
    persistentVolumeClaim:
      claimName: postgres-pvc
---
kind: PersistentVolumeClaim
spec:
  accessModes: [ReadWriteOnce]
  resources:
    requests:
      storage: 1Gi
\end{lstlisting}
\end{tcolorbox}

\vspace{6pt}
\textbf{Déploiement sur Minikube (local) :}
\begin{tcolorbox}[codebox]
\begin{lstlisting}
$ kubectl apply -f k8s/namespace.yaml
$ kubectl apply -f k8s/configmap.yaml
$ kubectl apply -f k8s/secret.yaml
$ kubectl apply -f k8s/postgres-deployment.yaml
$ kubectl apply -f k8s/backend-deployment.yaml
$ kubectl apply -f k8s/frontend-deployment.yaml
